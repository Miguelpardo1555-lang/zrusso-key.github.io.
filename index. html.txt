<script>
    // Generador matemático de Key diaria cambiante
    function generateDailyKey() {
        const d = new Date();
        const dateString = d.getFullYear() + "-" + (d.getMonth() + 1) + "-" + d.getDate();
        
        // Algoritmo simple de encriptado diario (Crea un código único por fecha)
        let hash = 0;
        for (let i = 0; i < dateString.length; i++) {
            hash = dateString.charCodeAt(i) + ((hash << 5) - hash);
        }
        
        // Convierte el hash en una palabra clave limpia (Ej: ZR-A1B2C3)
        const code = Math.abs(hash).toString(16).toUpperCase().substring(0, 6);
        return "ZR-" + code;
    }

    const currentKey = generateDailyKey();
    document.getElementById("keyDisplay").innerText = currentKey;

    // Botón de copiado automático para iPhone y Android
    document.getElementById("copyBtn").addEventListener("click", function() {
        navigator.clipboard.writeText(currentKey).then(() => {
            const btn = document.getElementById("copyBtn");
            btn.innerText = "¡COPIADO!";
            btn.style.backgroundColor = "#00ffb4";
            btn.style.color = "#0a0a0c";
            setTimeout(() => {
                btn.innerText = "COPIAR KEY";
                btn.style.backgroundColor = "#141a1a";
                btn.style.color = "#00ffb4";
            }, 2000);
        });
    });
